# There are several other ways to do this

f -> $n { say "Hi {$n}" };  # Explicit argument, no parenthesis

# instead of parenthesis
my $obj; ...;
$obj.f(-> $n { say "Hi {$n}" });  # Explicit argument
$obj.f: -> $n { say "Hi {$n}" };  # Explicit argument, no parenthesis
